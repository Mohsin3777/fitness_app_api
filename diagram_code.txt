//// Schema: fitness_app

Enum fitness_level {
  Beginner
  Intermediate
  Advanced
}

Enum goal_type {
  "Weight loss"
  "Gain muscle"
  "Improve fitness"
}

Table fitness_app.accounts {
  id int [pk, increment]
  first_name varchar
  last_name varchar
  phone varchar [unique, not null]
  email varchar [unique, not null]
  password varchar [not null]
  age int
  weight float
  goal_weight float
  fitness_level fitness_level
  your_goal goal_type
    category_id int [ref: > fitness_app.categories.id, not null]

  created_at timestamp [default: `now()`]

  Indexes {
    (email, phone) [name: "user_contact_index"]
  }
}
Table fitness_app.categories {
  id int [pk, increment]
  name varchar [unique, not null]

  Note: 'Available categories like Yoga, Gym, Stretch, Cardio'
}



Enum difficulty_level {
  Beginner
  Intermediate
  Advanced
}

Table fitness_app.exercises {
  id int [pk, increment]
  name varchar [not null, unique]
  description text
  category_id int [ref: > fitness_app.categories.id, not null]
  duration_minutes int [note: 'Average time in minutes'] // optional
  reps int [note: 'Optional if its a reps-based exercise']
  equipment varchar [note: 'Optional: dumbbell, mat, etc.']
  difficulty difficulty_level
  video_url varchar [note: 'Optional instructional video']
  created_at timestamp [default: `now()`]

  Indexes {
    (name, category_id) [unique]
  }
}
Table fitness_app.workout_plans {
  id int [pk, increment]
  name varchar [not null]
  description text
  category_id int [ref: > fitness_app.categories.id]
  created_by int [ref: > fitness_app.accounts.id]
  duration_weeks int
  difficulty difficulty_level
  created_at timestamp [default: `now()`]
}
Table fitness_app.workout_plan_exercises {
  id int [pk, increment]
  plan_id int [ref: > fitness_app.workout_plans.id]
  exercise_id int [ref: > fitness_app.exercises.id]
  day_of_week varchar [note: 'e.g., Monday']
  order_number int [note: 'Exercise order in workout']
  sets int
  reps int
}

  id int [pk, increment]

  created_at timestamp [default: `now()`]
